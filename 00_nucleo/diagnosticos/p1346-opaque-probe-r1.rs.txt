use std::fs::File;
use std::io::{self, Read};

const DOMAIN: &[u8] = b"P1345-OPAQUE-PROBE-CHALLENGE-V1\0";

fn sha256(input: &[u8]) -> [u8; 32] {
    const H0: [u32; 8] = [
        0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a, 0x510e527f, 0x9b05688c,
        0x1f83d9ab, 0x5be0cd19,
    ];
    const K: [u32; 64] = [
        0x428a2f98, 0x71374491, 0xb5c0fbcf, 0xe9b5dba5, 0x3956c25b, 0x59f111f1,
        0x923f82a4, 0xab1c5ed5, 0xd807aa98, 0x12835b01, 0x243185be, 0x550c7dc3,
        0x72be5d74, 0x80deb1fe, 0x9bdc06a7, 0xc19bf174, 0xe49b69c1, 0xefbe4786,
        0x0fc19dc6, 0x240ca1cc, 0x2de92c6f, 0x4a7484aa, 0x5cb0a9dc, 0x76f988da,
        0x983e5152, 0xa831c66d, 0xb00327c8, 0xbf597fc7, 0xc6e00bf3, 0xd5a79147,
        0x06ca6351, 0x14292967, 0x27b70a85, 0x2e1b2138, 0x4d2c6dfc, 0x53380d13,
        0x650a7354, 0x766a0abb, 0x81c2c92e, 0x92722c85, 0xa2bfe8a1, 0xa81a664b,
        0xc24b8b70, 0xc76c51a3, 0xd192e819, 0xd6990624, 0xf40e3585, 0x106aa070,
        0x19a4c116, 0x1e376c08, 0x2748774c, 0x34b0bcb5, 0x391c0cb3, 0x4ed8aa4a,
        0x5b9cca4f, 0x682e6ff3, 0x748f82ee, 0x78a5636f, 0x84c87814, 0x8cc70208,
        0x90befffa, 0xa4506ceb, 0xbef9a3f7, 0xc67178f2,
    ];
    let bit_len = (input.len() as u64) * 8;
    let mut data = input.to_vec();
    data.push(0x80);
    while data.len() % 64 != 56 {
        data.push(0);
    }
    data.extend_from_slice(&bit_len.to_be_bytes());
    let mut h = H0;
    for block in data.chunks_exact(64) {
        let mut w = [0u32; 64];
        for (i, chunk) in block.chunks_exact(4).enumerate() {
            w[i] = u32::from_be_bytes(chunk.try_into().unwrap());
        }
        for i in 16..64 {
            let s0 =
                w[i - 15].rotate_right(7) ^ w[i - 15].rotate_right(18) ^ (w[i - 15] >> 3);
            let s1 =
                w[i - 2].rotate_right(17) ^ w[i - 2].rotate_right(19) ^ (w[i - 2] >> 10);
            w[i] = w[i - 16].wrapping_add(s0).wrapping_add(w[i - 7]).wrapping_add(s1);
        }
        let (mut a, mut b, mut c, mut d, mut e, mut f, mut g, mut hh) =
            (h[0], h[1], h[2], h[3], h[4], h[5], h[6], h[7]);
        for i in 0..64 {
            let s1 = e.rotate_right(6) ^ e.rotate_right(11) ^ e.rotate_right(25);
            let ch = (e & f) ^ ((!e) & g);
            let t1 = hh
                .wrapping_add(s1)
                .wrapping_add(ch)
                .wrapping_add(K[i])
                .wrapping_add(w[i]);
            let s0 = a.rotate_right(2) ^ a.rotate_right(13) ^ a.rotate_right(22);
            let maj = (a & b) ^ (a & c) ^ (b & c);
            let t2 = s0.wrapping_add(maj);
            hh = g;
            g = f;
            f = e;
            e = d.wrapping_add(t1);
            d = c;
            c = b;
            b = a;
            a = t1.wrapping_add(t2);
        }
        for (slot, value) in h.iter_mut().zip([a, b, c, d, e, f, g, hh]) {
            *slot = slot.wrapping_add(value);
        }
    }
    let mut out = [0u8; 32];
    for (chunk, value) in out.chunks_exact_mut(4).zip(h) {
        chunk.copy_from_slice(&value.to_be_bytes());
    }
    out
}

fn hex(bytes: &[u8]) -> String {
    const D: &[u8; 16] = b"0123456789abcdef";
    let mut out = String::with_capacity(bytes.len() * 2);
    for &b in bytes {
        out.push(D[(b >> 4) as usize] as char);
        out.push(D[(b & 15) as usize] as char);
    }
    out
}

fn decode32(value: &str) -> Option<[u8; 32]> {
    if value.len() != 64
        || !value
            .bytes()
            .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
    {
        return None;
    }
    let mut out = [0u8; 32];
    for i in 0..32 {
        let d = |b: u8| if b <= b'9' { b - b'0' } else { b - b'a' + 10 };
        out[i] = (d(value.as_bytes()[2 * i]) << 4) | d(value.as_bytes()[2 * i + 1]);
    }
    Some(out)
}

fn request(raw: &str) -> Option<([u8; 32], [u8; 32])> {
    let prefix =
        "{\"schema\":\"p1345-opaque-probe-request-r2\",\"fresh_challenge_hex\":\"";
    let middle = "\",\"invocation_nonce_hex\":\"";
    let suffix = "\"}\n";
    let rest = raw.strip_prefix(prefix)?;
    let (challenge, rest) = rest.split_once(middle)?;
    let nonce = rest.strip_suffix(suffix)?;
    Some((decode32(challenge)?, decode32(nonce)?))
}

fn main() {
    let mut raw = String::new();
    io::stdin().read_to_string(&mut raw).unwrap();
    let (challenge, invocation) = request(&raw).expect("closed P1345 request");
    assert_ne!(challenge, invocation);
    let mut process = [0u8; 32];
    File::open("/dev/urandom").unwrap().read_exact(&mut process).unwrap();
    assert_ne!(process, challenge);
    assert_ne!(process, invocation);
    let process_digest = sha256(&process);
    let invocation_digest = sha256(&invocation);
    let mut response = Vec::with_capacity(DOMAIN.len() + 64);
    response.extend_from_slice(DOMAIN);
    response.extend_from_slice(&challenge);
    response.extend_from_slice(&process);
    let response_digest = sha256(&response);
    let projection_digest = sha256(b"p1345:opaque-dictionary-public-projection:v1");
    println!("{{\"schema\":\"p1345-opaque-probe-primitive-output-r2\",\"challenge_response_sha256\":\"{}\",\"invocation_nonce_sha256\":\"{}\",\"process_nonce_hex\":\"{}\",\"process_nonce_sha256\":\"{}\",\"opaque_handle_count\":1,\"public_projection_sha256\":\"{}\",\"payload_octets_exposed\":0,\"completed_phase\":\"OPAQUE_PAIR_PROJECTED\"}}",hex(&response_digest),hex(&invocation_digest),hex(&process),hex(&process_digest),hex(&projection_digest));
}
